// 2. WEB AUDIO SYNTHESIZER (MUZIK LATAR CERIA & KESAN BUNYI)
// ============================================================================
class SoundEngine {
  constructor() {
    this.ctx = null;
    this.enabled = true;
    this.bgmInterval = null;
    this.noteIndex = 0;
    // Melodi ceria kanak-kanak (Pentatonic Major C)
    this.melody = [
      523.25, 659.25, 783.99, 659.25, // C5 E5 G5 E5
      587.33, 698.46, 880.00, 698.46, // D5 F5 A5 F5
      523.25, 659.25, 783.99, 1046.50, // C5 E5 G5 C6
      783.99, 659.25, 587.33, 523.25  // G5 E5 D5 C5
    ];
  }

  init() {
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) {
        this.ctx = new AudioCtx();
      }
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume();
    }
  }

  setEnabled(val) {
    this.enabled = val;
    if (!val) {
      this.stopBGM();
    } else {
      this.startBGM();
    }
  }

  playTone(freq, duration = 0.2, type = 'sine', volume = 0.12, delay = 0) {
    if (!this.enabled) return;
    this.init();
    if (!this.ctx) return;

    const now = this.ctx.currentTime + delay;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = type;
    osc.frequency.setValueAtTime(freq, now);

    gain.gain.setValueAtTime(0.001, now);
    gain.gain.exponentialRampToValueAtTime(volume, now + 0.02);
    gain.gain.exponentialRampToValueAtTime(0.001, now + duration);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + duration + 0.02);
  }

  startBGM() {
    if (!this.enabled || this.bgmInterval) return;
    this.init();
    this.bgmInterval = setInterval(() => {
      if (!this.enabled) return;
      const freq = this.melody[this.noteIndex % this.melody.length];
      this.playTone(freq, 0.28, 'triangle', 0.045, 0);
      this.noteIndex++;
    }, 450);
  }

  stopBGM() {
    if (this.bgmInterval) {
      clearInterval(this.bgmInterval);
      this.bgmInterval = null;
    }
  }

  playTick(progress) {
    // Nada meningkat mengikut peratus tahanan kepala
    const pitch = 440 + progress * 440;
    this.playTone(pitch, 0.08, 'sine', 0.07, 0);
  }

  playCorrect() {
    // Bunyi kejayaan ceria (Arpeggio C Major naik)
    this.playTone(523.25, 0.15, 'triangle', 0.18, 0);
    this.playTone(659.25, 0.15, 'triangle', 0.18, 0.1);
    this.playTone(783.99, 0.15, 'triangle', 0.18, 0.2);
    this.playTone(1046.50, 0.45, 'triangle', 0.22, 0.3);
  }

  playIncorrect() {
    // Bunyi lembut "Cuba Lagi" (tidak menakutkan kanak-kanak)
    this.playTone(392.00, 0.2, 'sine', 0.15, 0);
    this.playTone(329.63, 0.35, 'sine', 0.15, 0.18);
  }

  playVictory() {
    // Fanfare tamat misi
    const notes = [523.25, 659.25, 783.99, 1046.50, 783.99, 1046.50];
    const times = [0, 0.12, 0.24, 0.36, 0.55, 0.7];
    notes.forEach((n, idx) => {
      this.playTone(n, idx === notes.length - 1 ? 0.6 : 0.18, 'triangle', 0.2, times[idx]);
    });
  }
}

const ConfettiCelebration = ({ active }) => {
  if (!active) return null;
  const colors = ['#F43F5E', '#FBBF24', '#34D399', '#38BDF8', '#A78BFA', '#FB923C'];
  const items = Array.from({ length: 36 });

  return (
    <div className="fixed inset-0 pointer-events-none z-50 overflow-hidden">
      {items.map((_, idx) => {
        const left = `${(idx * 2.8) % 100}%`;
        const delay = `${(idx % 8) * 0.12}s`;
        const duration = `${1.8 + (idx % 5) * 0.35}s`;
        const color = colors[idx % colors.length];
        const isStar = idx % 3 === 0;

        return (
          <div
            key={idx}
            className="absolute -top-8 animate-[confettiFall_linear_forwards]"
            style={{
              left,
              animationDelay: delay,
              animationDuration: duration
            }}
          >
            {isStar ? (
              <span className="text-2xl">⭐</span>
            ) : (
              <div
                className="w-3.5 h-3.5 rounded-sm"
                style={{
                  backgroundColor: color,
                  transform: `rotate(${idx * 37}deg)`
                }}
              />
            )}
          </div>
        );
      })}
    </div>
  );
};

// 5. KOMPONEN PENUNJUK KEMAJUAN BULAT (CIRCULAR HOLD PROGRESS)
// ============================================================================
const CircularProgressRing = ({ progress = 0, size = 76, strokeWidth = 8, color = '#10B981' }) => {
  const radius = (size - strokeWidth) / 2;
  const circumference = radius * 2 * Math.PI;
  const offset = circumference - Math.min(1, Math.max(0, progress)) * circumference;

  return (
    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
      <svg width={size} height={size} className="-rotate-90 drop-shadow">
        <circle
          cx={size / 2}
          cy={size / 2}
          r={radius}
          fill="rgba(255,255,255,0.9)"
          stroke="#E5E7EB"
          strokeWidth={strokeWidth}
        />
        <circle
          cx={size / 2}
          cy={size / 2}
          r={radius}
          fill="transparent"
          stroke={color}
          strokeWidth={strokeWidth}
          strokeDasharray={circumference}
          strokeDashoffset={offset}
          strokeLinecap="round"
          className="transition-all duration-75"
        />
      </svg>
      <span className="absolute text-sm font-black text-slate-800">
        {Math.round(progress * 100)}%
      </span>
    </div>
  );
};

      if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
        throw new Error('Pelayar ini tidak menyokong akses kamera.');
      }

      // Hentikan strim lama jika ada
      if (streamRef.current) {
        streamRef.current.getTracks().forEach((track) => track.stop());
      }

      const stream = await navigator.mediaDevices.getUserMedia({
        video: {
          facingMode: 'user',
          width: { ideal: 640 },
          height: { ideal: 480 }
        },
        audio: false
      });

      streamRef.current = stream;

      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        await videoRef.current.play();
      }

      setCameraStatus('LOADING_MODEL');

      // Muatkan MediaPipe Tasks Vision melalui skrip modul global supaya serasi dengan semua bundler
      if (!faceLandmarkerRef.current) {
        if (!window.__mediapipeVision) {
          await new Promise((resolve, reject) => {
            const script = document.createElement('script');
            script.type = 'module';
            script.textContent = `
              import * as vision from 'https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/vision_bundle.mjs';
              window.__mediapipeVision = vision;
              window.dispatchEvent(new Event('mediapipe-vision-loaded'));
            `;
            const onLoaded = () => {
              window.removeEventListener('mediapipe-vision-loaded', onLoaded);
              resolve();
            };
            window.addEventListener('mediapipe-vision-loaded', onLoaded);
            script.onerror = () => reject(new Error('Gagal memuatkan pustaka pengesan wajah.'));
            document.head.appendChild(script);
            setTimeout(() => {
              if (!window.__mediapipeVision) reject(new Error('Masa tamat memuatkan model wajah.'));
            }, 12000);
          });
        }

        const { FaceLandmarker, FilesetResolver } = window.__mediapipeVision;

        const filesetResolver = await FilesetResolver.forVisionTasks(
          'https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@0.10.14/wasm'
        );

        // Elakkan log INFO dalaman TensorFlow Lite (XNNPACK) daripada dicetak sebagai console.error
        const originalConsoleError = console.error;
        console.error = (...args) => {
          if (typeof args[0] === 'string' && args[0].includes('TensorFlow Lite XNNPACK delegate')) {
            return;
          }
          originalConsoleError.apply(console, args);
        };

        try {
          faceLandmarkerRef.current = await FaceLandmarker.createFromOptions(filesetResolver, {
            baseOptions: {
              modelAssetPath:
                'https://storage.googleapis.com/mediapipe-models/face_landmarker/face_landmarker/float16/1/face_landmarker.task',
              delegate: 'CPU'
            },
            outputFaceBlendshapes: false,
            runningMode: 'VIDEO',
            numFaces: 1
          });
        } finally {
          console.error = originalConsoleError;
        }
      }

      setCameraStatus('READY');
      startDetectionLoop();
    } catch (err) {
      let msg = 'Kamera tidak dapat diakses. Jangan risau, kamu masih boleh bermain menggunakan butang sentuh atau kekunci anak panah!';
      if (err?.name === 'NotAllowedError' || err?.name === 'PermissionDeniedError') {
        msg = 'Kebenaran kamera ditolak atau disekat oleh tetingkap pratonton. Klik "MULA (MOD SENTUH / KEKUNCI)" di bawah untuk bermain!';
      } else if (err?.name === 'NotFoundError') {
        msg = 'Tiada kamera dikesan pada peranti ini. Gunakan Mod Sentuh / Kekunci untuk bermain!';
      }
      setCameraErrorMsg(msg);
      setCameraStatus('ERROR');
    }
  };

  // Sambungkan semula video element apabila bertukar antara skrin SETUP dan PLAYING
  useEffect(() => {
    if (videoRef.current && streamRef.current && videoRef.current.srcObject !== streamRef.current) {
      videoRef.current.srcObject = streamRef.current;
      videoRef.current.play().catch(() => {});
    }
  }, [screen, cameraStatus]);

  // Bersihkan kamera & loop apabila komponen ditutup
  useEffect(() => {
    return () => {
      if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);
      if (streamRef.current) {
        streamRef.current.getTracks().forEach((t) => t.stop());
      }
      soundEngine.stopBGM();
    };
  }, []);

  // ==========================================================================
  // PENGESANAN SUDUT SENGETAN KEPALA & LOGIK TAHAN 0.8 SAAT
  // ==========================================================================
  const TILT_THRESHOLD_DEG = 15; // Ambang 15 darjah
  const HOLD_DURATION_MS = 800;  // Tahan selama 0.8 saat

  const handleAnswerSubmission = useCallback(
    (selectedSide) => {
      if (feedbackRef.current || cooldownRef.current) return;

      const currentQ = QUESTIONS[currentIndex];
      if (!currentQ) return;

      // Reset pemegang masa
      holdStartTimeRef.current = null;
      holdingSideRef.current = null;
      setHoldProgress(0);
      setActiveTiltSide(null);

      // Aktifkan tempoh bertenang (cooldown) 1.2 saat
      setCooldownActive(true);
      setTimeout(() => {
        setCooldownActive(false);
      }, 1200);

      if (selectedSide === currentQ.correctSide) {
        soundEngine.playCorrect();
        if (!questionSolved) {
          setScore((prev) => prev + 10);
          setCorrectCount((prev) => prev + 1);
          setQuestionSolved(true);
        }
        setFeedback({
          type: 'CORRECT',
          title: 'HEBAT! JAWAPAN BETUL! 🌟',
          explanation: currentQ.explanation
        });
      } else {
        soundEngine.playIncorrect();
        setFeedback({
          type: 'INCORRECT',
          title: 'CUBA LAGI! 💪',
          explanation: 'Fikirkan semula kesan baik mengamalkan adab makan dan minum. Kamu pasti boleh!'
        });
      }
    },
    [currentIndex, questionSolved]
  );

  // Simpan rujukan fungsi terkini untuk digunakan dalam animation frame
  const handleAnswerSubmissionRef = useRef(handleAnswerSubmission);
  useEffect(() => {
    handleAnswerSubmissionRef.current = handleAnswerSubmission;
  }, [handleAnswerSubmission]);

  const startDetectionLoop = () => {
    if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);

    const detect = () => {
      const video = videoRef.current;
      const landmarker = faceLandmarkerRef.current;

      let detectedAngle = null;

      if (video && landmarker && video.readyState >= 2) {
        try {
          const nowInMs = performance.now();
          if (video.currentTime !== lastVideoTimeRef.current) {
            lastVideoTimeRef.current = video.currentTime;
            const results = landmarker.detectForVideo(video, nowInMs);

            if (results && results.faceLandmarks && results.faceLandmarks.length > 0) {
              const landmarks = results.faceLandmarks[0];
              const eyeA = landmarks[33];
              const eyeB = landmarks[263];

              if (eyeA && eyeB) {
                const dx = eyeB.x - eyeA.x;
                const dy = eyeB.y - eyeA.y;
                const angleRad = Math.atan2(dy, dx);
                const angleDeg = -((angleRad * 180) / Math.PI);

                setFaceDetected(true);
                setRawRollAngle(angleDeg);
                detectedAngle = angleDeg - calibrationOffsetRef.current;
              }
            } else {
              setFaceDetected(false);
            }
          }
        } catch (frameErr) {
          // Abaikan ralat bingkai sementara supaya gelung animasi tidak terhenti
        }
      }

      // Kawalan Alternatif Papan Kekunci (Jika kekunci anak panah ditekan)
      if (keyHoldSideRef.current === 'LEFT') {
        detectedAngle = -22;
      } else if (keyHoldSideRef.current === 'RIGHT') {
        detectedAngle = 22;
      }

      // Kemaskini logik pemilihan jawapan semasa di Skrin Permainan Utama
      if (
        screenRef.current === 'PLAYING' &&
        !feedbackRef.current &&
        !cooldownRef.current &&
        detectedAngle !== null
      ) {
        let currentSide = null;
        if (detectedAngle <= -TILT_THRESHOLD_DEG) {
          currentSide = 'LEFT';
        } else if (detectedAngle >= TILT_THRESHOLD_DEG) {
          currentSide = 'RIGHT';
        }

        if (currentSide) {
          setActiveTiltSide(currentSide);
          if (holdingSideRef.current !== currentSide) {
            holdingSideRef.current = currentSide;
            holdStartTimeRef.current = performance.now();
            lastTickStepRef.current = 0;
          } else {
            const elapsed = performance.now() - holdStartTimeRef.current;
            const progress = Math.min(1, elapsed / HOLD_DURATION_MS);
            setHoldProgress(progress);

            // Bunyi detik lembut setiap 25% kemajuan
            const step = Math.floor(progress * 4);
            if (step > lastTickStepRef.current && step < 4) {
              lastTickStepRef.current = step;
              soundEngine.playTick(progress);
            }

            if (progress >= 1) {
              handleAnswerSubmissionRef.current(currentSide);
            }
          }
        } else {
          // Kepala kembali tegak sebelum 0.8 saat
          if (holdingSideRef.current !== null) {
            holdingSideRef.current = null;
            holdStartTimeRef.current = null;
            setActiveTiltSide(null);
            setHoldProgress(0);
          }
        }
      } else if (screenRef.current !== 'PLAYING' || feedbackRef.current || cooldownRef.current) {
        if (holdingSideRef.current !== null) {
          holdingSideRef.current = null;
          holdStartTimeRef.current = null;
          setActiveTiltSide(null);
          setHoldProgress(0);
        }
      }

      animFrameRef.current = requestAnimationFrame(detect);
    };

    animFrameRef.current = requestAnimationFrame(detect);
  };

  // Pastikan loop berjalan walaupun dalam mod tanpa kamera (supaya kekunci anak panah berfungsi lancar)
  useEffect(() => {
    if (screen === 'PLAYING' && !animFrameRef.current) {
      startDetectionLoop();
    }
  }, [screen]);

  // ==========================================================================
  // KAWALAN ALTERNATIF: PAPAN KEKUNCI (ARROW LEFT / ARROW RIGHT)
  // ==========================================================================
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (screenRef.current !== 'PLAYING' || feedbackRef.current || cooldownRef.current) return;
      if (e.key === 'ArrowLeft') {
        e.preventDefault();
        keyHoldSideRef.current = 'LEFT';
        setRawRollAngle(calibrationOffsetRef.current - 22);
      } else if (e.key === 'ArrowRight') {
        e.preventDefault();
        keyHoldSideRef.current = 'RIGHT';
        setRawRollAngle(calibrationOffsetRef.current + 22);
      }
    };

    const handleKeyUp = (e) => {
      if (e.key === 'ArrowLeft' || e.key === 'ArrowRight') {
        keyHoldSideRef.current = null;
        if (!faceDetected) {
          setRawRollAngle(calibrationOffsetRef.current);
        }
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    window.addEventListener('keyup', handleKeyUp);
    return () => {
      window.removeEventListener('keydown', handleKeyDown);
      window.removeEventListener('keyup', handleKeyUp);
    };
  }, [faceDetected]);

  // ==========================================================================
  // FUNGSI NAVIGASI & KALIBRASI
  // ==========================================================================
  const handleCalibrate = () => {
    setCalibrationOffset(rawRollAngle);
    setCalibratedBanner(true);
    soundEngine.playTone(660, 0.15, 'sine', 0.12);
    setTimeout(() => setCalibratedBanner(false), 2200);
  };

  const startNewGame = () => {
    soundEngine.init();
    if (soundOn) soundEngine.startBGM();
    setCurrentIndex(0);
    setScore(0);
    setCorrectCount(0);
    setQuestionSolved(false);
    setFeedback(null);
    setActiveTiltSide(null);
    setHoldProgress(0);
    setCooldownActive(true);
    setTimeout(() => setCooldownActive(false), 1000);
    setScreen('PLAYING');
  };

  const goToNextQuestion = () => {
    setFeedback(null);
    setQuestionSolved(false);
    setActiveTiltSide(null);
    setHoldProgress(0);
    setCooldownActive(true);
    setTimeout(() => setCooldownActive(false), 1100);

    if (currentIndex + 1 < QUESTIONS.length) {
      setCurrentIndex((prev) => prev + 1);
    } else {
      setScreen('CELEBRATION');
      soundEngine.playVictory();
    }
  };

  const retryCurrentQuestion = () => {
    setFeedback(null);
    setActiveTiltSide(null);
    setHoldProgress(0);
    setCooldownActive(true);
    setTimeout(() => setCooldownActive(false), 1000);
  };

  const currentQ = QUESTIONS[currentIndex];

  // ==========================================================================
  // RENDER ANTARA MUKA (4 SKRIN UTAMA)
  // ==========================================================================
  return (
    <div className="min-h-screen w-full overflow-x-hidden relative font-['Nunito',sans-serif] text-slate-800 select-none bg-gradient-to-b from-[#E0F4FF] via-[#E8FDF5] to-[#FFF6CC]">
      {/* Google Fonts & Animasi Kustom */}
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Fredoka:wght@500;600;700&family=Nunito:wght@600;700;800;900&display=swap');
        
        .font-display {
          font-family: 'Fredoka', cursive, sans-serif;
        }
        @keyframes float {
          0%, 100% { transform: translateY(0px); }
          50% { transform: translateY(-8px); }
        }
        @keyframes drift {
          0% { transform: translateX(-10px); }
          50% { transform: translateX(15px); }
          100% { transform: translateX(-10px); }
        }
        @keyframes confettiFall {
          0% { transform: translateY(-20px) rotate(0deg); opacity: 1; }
          100% { transform: translateY(105vh) rotate(720deg); opacity: 0.85; }
        }
      `}</style>

      {/* ==================== LATAR BELAKANG TAMAN & BILIK DARJAH KARTUN ==================== */}
      <div className="fixed inset-0 pointer-events-none overflow-hidden z-0">
        {/* Matahari Ceria */}
        <div className="absolute top-6 right-8 w-24 h-24 rounded-full bg-yellow-300/80 blur-sm animate-pulse" />
        <div className="absolute top-8 right-10 w-20 h-20 rounded-full bg-amber-300 border-4 border-yellow-100 shadow-lg flex items-center justify-center text-3xl">
          ☀️
        </div>

        {/* Awan Gebu */}
        <div className="absolute top-12 left-10 text-6xl opacity-75 animate-[drift_9s_ease-in-out_infinite]">☁️</div>
        <div className="absolute top-24 left-1/3 text-5xl opacity-60 animate-[drift_12s_ease-in-out_infinite]">☁️</div>
        <div className="absolute top-16 right-1/4 text-6xl opacity-70 animate-[drift_10s_ease-in-out_infinite]">☁️</div>

        {/* Bintang Terapung Pastel */}
        <div className="absolute top-1/4 left-6 text-2xl opacity-60 animate-bounce">✨</div>
        <div className="absolute top-1/3 right-8 text-2xl opacity-60 animate-bounce">⭐</div>

        {/* Bukit Hijau & Bunga Taman di Bahagian Bawah */}
        <svg
          className="absolute bottom-0 left-0 w-full h-36 md:h-48"
          viewBox="0 0 1440 220"
          preserveAspectRatio="none"
        >
          <path
            d="M0,110 C320,30 640,170 960,90 C1200,30 1350,100 1440,80 L1440,220 L0,220 Z"
            fill="#A7F3D0"
          />
          <path
            d="M0,150 C420,90 880,195 1440,120 L1440,220 L0,220 Z"
            fill="#6EE7B7"
          />
        </svg>
        <div className="hidden sm:flex fixed bottom-3 left-8 gap-6 text-2xl opacity-85">
          <span>🌷</span>
          <span>🌼</span>
          <span>🌻</span>
        </div>
      </div>

      {/* ==================== BAR ATAS GLOBAL (BUNYI & PENUNJUK SUBJEK) ==================== */}
      <header className="relative z-20 max-w-6xl mx-auto px-4 pt-3 pb-2 flex items-center justify-between gap-2">
        <div className="flex items-center gap-2 bg-white/90 backdrop-blur px-3.5 py-1.5 rounded-full border-2 border-sky-200 shadow-sm">
          <span className="text-lg">🕌</span>
          <span className="text-xs sm:text-sm font-extrabold text-sky-800 tracking-wide">
            Pendidikan Islam • Sekolah Rendah
          </span>
        </div>

        <div className="flex items-center gap-2">
          {screen === 'PLAYING' && (
            <button
              onClick={() => setScreen('SETUP')}
              className="flex items-center gap-1.5 bg-white/95 hover:bg-amber-50 text-amber-800 font-extrabold text-xs sm:text-sm px-3.5 py-2 rounded-full border-2 border-amber-300 shadow-sm active:scale-95 transition"
              title="Kalibrasi semula kedudukan kepala"
            >
              <Compass className="w-4 h-4 text-amber-600" />
              <span className="hidden sm:inline">Kalibrasi</span>
            </button>
          )}

          <button
            onClick={toggleSound}
            className={`flex items-center gap-1.5 font-extrabold text-xs sm:text-sm px-3.5 py-2 rounded-full border-2 shadow-sm active:scale-95 transition ${
              soundOn
                ? 'bg-emerald-100 border-emerald-400 text-emerald-900'
                : 'bg-rose-100 border-rose-300 text-rose-800'
            }`}
            aria-label="Tukar Bunyi"
          >
            {soundOn ? <Volume2 className="w-4 h-4" /> : <VolumeX className="w-4 h-4" />}
            <span>{soundOn ? 'Bunyi: ON' : 'Bunyi: OFF'}</span>
          </button>
        </div>
      </header>

      {/* ==================== ELEMEN VIDEO TERSEMBUNYI / DIKONGSI UNTUK PENGESANAN ==================== */}
      {/* Sentiasa wujud dalam DOM supaya aliran kamera tidak terputus antara skrin */}
      <div
        className={
          screen === 'PLAYING'
            ? 'fixed bottom-4 right-4 z-30 flex flex-col items-end pointer-events-none'
            : 'hidden'
        }
      >
        {cameraStatus === 'READY' && (
          <div className="bg-white/95 p-2 rounded-2xl border-4 border-sky-300 shadow-xl pointer-events-auto">
            <div className="relative w-32 h-24 sm:w-40 sm:h-28 rounded-xl overflow-hidden bg-slate-900">
              <video
                ref={screen === 'PLAYING' ? videoRef : null}
                playsInline
                muted
                className="w-full h-full object-cover -scale-x-100"
              />
              {/* Garis Panduan Tegak & Penunjuk Sudut */}
              <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
                <div
                  className="w-0.5 h-full bg-yellow-300/80 transition-transform duration-75"
                  style={{ transform: `rotate(${effectiveAngle}deg)` }}
                />
              </div>
              <div
                className={`absolute top-1.5 left-1.5 px-2 py-0.5 rounded-full text-[10px] font-extrabold ${
                  faceDetected ? 'bg-emerald-500 text-white' : 'bg-rose-500 text-white'
                }`}
              >
                {faceDetected ? `${Math.round(effectiveAngle)}°` : 'Cari Wajah...'}
              </div>
            </div>
            <div className="mt-1 flex items-center justify-between text-[11px] font-extrabold text-slate-600 px-1">
              <span>⬅️ Kiri</span>
              <span className="text-sky-700">Kamera</span>
              <span>Kanan ➡️</span>
            </div>
          </div>
        )}
      </div>

      {/* ===================================================================== */}
      {/* SKRIN 1: WELCOME SCREEN                                               */}
      {/* ===================================================================== */}
      {screen === 'WELCOME' && (
        <main className="relative z-10 max-w-4xl mx-auto px-4 py-4 sm:py-8 flex flex-col items-center text-center">
          {/* Lencana Topik */}
          <div className="inline-flex items-center gap-2 bg-[#FFE4EC] border-2 border-pink-300 text-pink-900 font-extrabold px-5 py-1.5 rounded-full text-sm sm:text-base shadow-sm mb-4">
            <Sparkles className="w-4 h-4 text-pink-600" />
            <span>Topik: Kesan Mengamalkan Adab Makan dan Minum</span>
          </div>

          {/* Tajuk Utama */}
          <h1 className="font-display text-3xl sm:text-5xl md:text-6xl font-bold text-sky-900 drop-shadow-sm leading-tight">
            🎮 MISI ADAB MAKAN &amp; MINUM
          </h1>

          {/* Subtajuk */}
          <p className="mt-3 font-display text-xl sm:text-2xl md:text-3xl font-semibold text-emerald-700 bg-white/80 px-6 py-2 rounded-2xl border-2 border-emerald-200 shadow-sm">
            “Sengetkan Kepala, Pilih Jawapan!”
          </p>

          {/* Watak Kartun Muslim & Muslimah */}
          <div className="my-6 sm:my-8 flex items-end justify-center gap-6 sm:gap-12 bg-white/75 backdrop-blur-sm px-8 py-6 rounded-3xl border-4 border-white shadow-xl">
            <CartoonCharacter gender="boy" name="Ali & Rakan" tiltAngle={-8} mood="happy" size={145} />
            <div className="hidden sm:flex flex-col items-center justify-center pb-8">
              <span className="text-4xl animate-bounce">🍽️</span>
              <span className="mt-2 text-xs font-extrabold text-amber-800 bg-amber-100 px-3 py-1 rounded-full">
                Bismillah!
              </span>
            </div>
            <CartoonCharacter gender="girl" name="Aisyah & Rakan" tiltAngle={8} mood="happy" size={145} />
          </div>

          {/* Butang Tindakan Utama */}
          <div className="flex flex-col sm:flex-row items-center justify-center gap-4 w-full max-w-lg">
            <button
              onClick={() => {
                soundEngine.init();
                if (soundOn) soundEngine.startBGM();
                setScreen('SETUP');
              }}
              className="w-full sm:w-auto flex-1 flex items-center justify-center gap-3 bg-gradient-to-b from-emerald-400 to-emerald-600 hover:from-emerald-500 hover:to-emerald-700 text-white font-display text-xl sm:text-2xl font-bold px-8 py-4 rounded-3xl border-b-4 border-emerald-800 shadow-lg hover:scale-105 active:scale-95 transition cursor-pointer"
            >
              <Play className="w-7 h-7 fill-white" />
              <span>MULA BERMAIN</span>
            </button>

            <button
              onClick={() => {
                soundEngine.init();
                setShowHowToModal(true);
              }}
              className="w-full sm:w-auto flex items-center justify-center gap-2 bg-gradient-to-b from-amber-300 to-amber-400 hover:from-amber-400 hover:to-amber-500 text-amber-950 font-display text-lg sm:text-xl font-bold px-6 py-4 rounded-3xl border-b-4 border-amber-600 shadow-lg hover:scale-105 active:scale-95 transition cursor-pointer"
            >
              <HelpCircle className="w-6 h-6" />
              <span>CARA BERMAIN</span>
            </button>
          </div>

          {/* Nota Privasi Mesra Kanak-kanak */}
          <p className="mt-6 flex items-center gap-2 text-xs sm:text-sm font-bold text-slate-600 bg-white/80 px-4 py-2 rounded-full">
            <ShieldCheck className="w-4 h-4 text-emerald-600 shrink-0" />
            <span>Selamat &amp; peribadi: Video kamera diproses terus di peranti kamu dan tidak disimpan.</span>
          </p>
        </main>
      )}

      {/* ===================================================================== */}
      {/* MODAL CARA BERMAIN                                                    */}
      {/* ===================================================================== */}
      {showHowToModal && (
        <div className="fixed inset-0 z-50 bg-slate-900/50 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white rounded-3xl border-4 border-sky-300 max-w-xl w-full p-6 sm:p-8 shadow-2xl relative">
            <h2 className="font-display text-2xl sm:text-3xl font-bold text-sky-900 text-center mb-4">
              📖 CARA BERMAIN
            </h2>

            <div className="space-y-3 text-base sm:text-lg font-bold text-slate-700">
              <div className="flex items-start gap-3 bg-sky-50 p-3.5 rounded-2xl border-2 border-sky-200">
                <span className="text-2xl">1️⃣</span>
                <p>Aktifkan kamera dan pastikan wajah kamu berada di tengah skrin.</p>
              </div>
              <div className="flex items-start gap-3 bg-emerald-50 p-3.5 rounded-2xl border-2 border-emerald-200">
                <span className="text-2xl">2️⃣</span>
                <p>
                  <span className="text-emerald-700 font-black">Sengetkan kepala ke KIRI</span> untuk memilih kad jawapan KIRI, atau <span className="text-pink-600 font-black">sengetkan kepala ke KANAN</span> untuk memilih kad jawapan KANAN.
                </p>
              </div>
              <div className="flex items-start gap-3 bg-amber-50 p-3.5 rounded-2xl border-2 border-amber-200">
                <span className="text-2xl">3️⃣</span>
